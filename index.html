const express = require('express'); const cors = require('cors'); const jwt = require('jsonwebtoken'); const multer = require('multer'); const nodemailer = require('nodemailer'); // Para emails de registro const app = express(); const PORT = 3000; // Simulación de DB (en producción, usa MongoDB) let users = []; let books = []; let purchases = []; // Middleware app.use(cors()); app.use(express.json()); app.use(express.static('public')); // Para servir archivos estáticos // Configuración de Multer para subir PDFs (simulado) const upload = multer({ dest: 'uploads/' }); // Autenticación app.post('/api/login', (req, res) => { const { username, password } = req.body; if (username === 'Lectuverso2026' && password === 'Lectuverso2026') { const token = jwt.sign({ isAdmin: true }, 'secretKey'); res.json({ token, isAdmin: true }); } else { const user = users.find(u => u.username === username && u.password === password); if (user) { const token = jwt.sign({ userId: user.id, isAdmin: false }, 'secretKey'); res.json({ token, isAdmin: false }); } else { res.status(401).json({ message: 'Credenciales incorrectas' }); } } }); // Registro (envía email al admin) app.post('/api/register', (req, res) => { const { name, email } = req.body; users.push({ id: Date.now(), name, email, password: 'temp' }); // Simulado // Enviar email (configura tu SMTP real) const transporter = nodemailer.createTransporter({ service: 'gmail', auth: { user: 'tuemail@gmail.com', pass: 'tupassword' } // Configura }); transporter.sendMail({ from: 'tuemail@gmail.com', to: 'magiccubicdesing@gmail.com', subject: 'Nuevo Registro', text: `Nuevo usuario: ${name}, ${email}` }); res.json({ message: 'Registro enviado' }); }); // Subir libro (admin) app.post('/api/upload-book', upload.single('pdf'), (req, res) => { books.push({ id: Date.now(), name: req.body.name, file: req.file.path }); res.json({ message: 'Libro subido' }); }); // Obtener libros app.get('/api/books', (req, res) => { res.json(books); }); // Comprar libro app.post('/api/purchase', (req, res) => { const { userId, bookId, amount } = req.body; purchases.push({ userId, bookId, amount, profitMargin: amount * 0.2 }); // Simulado res.json({ message: 'Compra realizada' }); }); // CRM (solo admin) app.get('/api/crm', (req, res) => { res.json({ users, purchases }); }); app.listen(PORT, () => console.log(`Servidor corriendo en http://localhost:${PORT}`));
